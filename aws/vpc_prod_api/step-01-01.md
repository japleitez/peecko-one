# Create VPC

aws ec2 create-vpc --cidr-block 10.0.0.0/16 --tag-specifications 'ResourceType=vpc,Tags=[{Key=Name,Value=prod-peecko-vpc}]'

# retrieve VPC ID

aws ec2 describe-vpcs --query 'Vpcs[*].{VpcId:VpcId,CIDR:CidrBlock,State:State,Name:Tags[?Key==`Name`].Value|[0]||`None`}'

# Note the VPC ID

$env:VPC_ID="vpc-03e69010829f69220"

# Enable DNS hostname resolution

aws ec2 modify-vpc-attribute --vpc-id "$env:VPC_ID" --enable-dns-hostnames
aws ec2 modify-vpc-attribute --vpc-id "$env:VPC_ID" --enable-dns-support
