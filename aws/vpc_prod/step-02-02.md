# ALB: Accept HTTP (80) and HTTPS (443) from internet

aws ec2 authorize-security-group-ingress --group-id "$env:ALB_SG" --protocol tcp --port 80 --cidr 0.0.0.0/0

aws ec2 authorize-security-group-ingress --group-id "$env:ALB_SG" --protocol tcp --port 443 --cidr 0.0.0.0/0
