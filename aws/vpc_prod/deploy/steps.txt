# 1. Reload systemd to pick up the new service file
sudo systemctl daemon-reload

# 2. Start the service
sudo systemctl start admin

# 3. Enable it to auto-start on reboot
sudo systemctl enable admin

# 4. Check it's running correctly
sudo systemctl status admin

# 5. If it failed, check the logs, this shows the last 50 lines of the service logs
