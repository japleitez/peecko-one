# Public Route Table

aws ec2 create-route-table --vpc-id "$env:VPC_ID" --tag-specifications 'ResourceType=route-table,Tags=[{Key=Name,Value=public-peecko-rt}]'
$env:PUBLIC_RT="rtb-04f6798903bf6a38f"

# Route public traffic to IGW

aws ec2 create-route --route-table-id "$env:PUBLIC_RT" --destination-cidr-block 0.0.0.0/0 --gateway-id "$env:IGW_ID"

# Associate public subnets with public route table

aws ec2 associate-route-table --subnet-id "$env:PUBLIC_SUBNET_1A" --route-table-id "$env:PUBLIC_RT"
aws ec2 associate-route-table --subnet-id "$env:PUBLIC_SUBNET_1B" --route-table-id "$env:PUBLIC_RT"

# Private Route Table

aws ec2 create-route-table --vpc-id "$env:VPC_ID" --tag-specifications 'ResourceType=route-table,Tags=[{Key=Name,Value=private-peecko-rt}]'
$env:PRIVATE_RT="rtb-08990426383859f88"

# Route private traffic to NAT Gateway

# Note: nat-gateway-id is obtained from the NAT GW creation

aws ec2 create-route --route-table-id "$env:PRIVATE_RT" --destination-cidr-block 0.0.0.0/0 --nat-gateway-id "$env:NAT_GW_ID"

# Associate private subnets with private route table

aws ec2 associate-route-table --subnet-id "$env:PRIVATE_SUBNET_1A" --route-table-id "$env:PRIVATE_RT"
aws ec2 associate-route-table --subnet-id "$env:PRIVATE_SUBNET_1B" --route-table-id "$env:PRIVATE_RT"

# Database Route Table (no outbound internet)

aws ec2 create-route-table --vpc-id "$env:VPC_ID" --tag-specifications 'ResourceType=route-table,Tags=[{Key=Name,Value=db-peecko-rt}]'
$env:DB_RT="rtb-01beec46618e0501e"

# Associate DB subnets (no route to IGW/NAT)

aws ec2 associate-route-table --subnet-id "$env:DB_SUBNET_1A" --route-table-id "$env:DB_RT"
aws ec2 associate-route-table --subnet-id "$env:DB_SUBNET_1B" --route-table-id "$env:DB_RT"
