# Create IAM role for EC2

aws iam create-role --role-name prod-peecko-ec2-app-role --assume-role-policy-document file://iam_prod_ec2_app_role.json --tags Key=Name,Value=prod-peecko-ec2-app-role

# Create instance profile

aws iam create-instance-profile --instance-profile-name prod-peecko-ec2-app-profile
aws iam add-role-to-instance-profile --instance-profile-name prod-peecko-ec2-app-profile --role-name prod-peecko-ec2-app-role

# Attach managed policies

aws iam attach-role-policy --role-name prod-peecko-ec2-app-role --policy-arn arn:aws:iam::aws:policy/CloudWatchAgentServerPolicy

# For Session Manager access

aws iam attach-role-policy --role-name prod-peecko-ec2-app-role --policy-arn arn:aws:iam::aws:policy/AmazonSSMManagedInstanceCore

# Create custom policy for Secrets Manager access

aws iam put-role-policy --role-name prod-peecko-ec2-app-role --policy-name prod-peecko-ec2-secrets-policy --policy-document file://iam_prod_ec2_app_secrets_policy.json

# Create policy for S3 backups

aws iam put-role-policy --role-name prod-peecko-ec2-app-role --policy-name prod-peecko-ec2-s3-backups --policy-document file://iam_prod_ec2_app_s3_backups_policy.json
